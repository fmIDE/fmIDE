/* fmJAML ( pOptionalInitialJSON ; pFmJAML )
   __                _         __  __ _      
  / _|              | |  /\   |  \/  | |     
 | |_ _ __ ___      | | /  \  | \  / | |     
 |  _| '_ ` _ \ _   | |/ /\ \ | |\/| | |     
 | | | | | | | | |__| / ____ \| |  | | |____ 
 |_| |_| |_| |_|\____/_/    \_\_|  |_|______|


`code from your keyboard`

**fmJAML** = **F**ile**M**aker **J**SON **A**s a **M**arkdown **L**angauge

…a Domain Specific Langauage (DSL) for encoding JSON as a simple, concise, readable string in FileMaker.

The fmJAML function takes an optional initial JSON and an fmJAML text as input 
and constructs the equivalent JSON, applying relevant processing logic to data type, value and path.

fmJAML, originally created to write fmIDE Action Scripts,
aims to be a generalised templating system for creating most any JSON.

@author: @mrwatson-de
@created: 2026-03-10
@version: See ¢release
@history: See fmJAML_changes

@param pOptionalInitialJSON (string) - a JSON that the fmJAML code builds upon or an empty string
@param pFmJAML (string) - the fmJAML code to express your JSON in FileMaker
@returns json (string) - the JSON constructed from the fmJAML string
@input $fmjaml_debug - fmLog357 log level to use: 0/""=none, 3=errors, 5=info, 7=debug (See fmLog357 for more).
@output $$fmjaml_log (string) - debugging log
@output $$fmjaml_debug_dtype (string) - just the data types

Note: Please do not use the reserved prefix `($)$fmjaml__` in your variables.
*/

/*******************
Process fmJAML lines
*******************/

While (
[
¢release = "2026-09-29 @mrwatson-de v0.36";

/* input param */
$fmjaml__json = pOptionalInitialJSON ;
¢lines = pFmJAML ;

/* input vars */
¢debug = Case( not IsEmpty ( $fmjaml_debug ) ; $fmjaml_debug ; Get ( ActiveModifierKeys ) ) ;
/* FIXME - Using the modifier keys for log level is a neat trick for quick debugging but a bit of a fudge. */

/* internal vars */
$fmjaml__path_prev = "" ;
¢mbs = Evaluate ( "MBS(\"Version\")" ) ≠ "?" ;
¢fmp_v_maj = 0 + GetValue ( Substitute ( Get ( ApplicationVersion ) ; "." ; ¶ ) ; 1 ) ;

/* output vars */
$$fmjaml_debug_dtype = "" ;
$$fmjaml_log = Case( ¢debug ≥ 6 ; "¢debug: " & ¢debug & ¶ & "¢mbs: " & ¢mbs & ¶ ) ;

/* Constants */
¢TAB = Char(9) ;
¢WS  = " " & ¢TAB ;
¢JSONDate      = 20 ;
¢JSONTime      = 21 ;
¢JSONTimestamp = 22 ;
¢JSONContainer = 23 ;

/* FOR EACH ¢line */

¢line_i = 1;
¢line_n = ValueCount ( ¢lines )

]; ¢line_i <= ¢line_n and Left( $fmjaml__json ; 1 ) ≠ "?" ;[

¢line  = GetValue ( ¢lines ; ¢line_i );
/* trim horizontally */
¢line  = Case( not Position( ¢line ; ¢TAB ; 1 ; 1 ) ; Trim( ¢line ) ; Let ([
  ¢tmp = Substitute( ¢line ; [" " ; ""] ; [¢TAB ; ""] ) ; 
  ¢ch1 = Left( ¢tmp ; 1 ) ; ¢chN = Right( ¢tmp ; 1 ) ; ¢pos = Position( ¢line ; ¢ch1 ; 1 ; 1 )
  ];
  Case( ¢pos ; Middle ( ¢line ; ¢pos ; Position( ¢line ; ¢chN ; Length ( ¢line ) ; -1 ) - ¢pos + 1 ) )
  ));

/* Split line into «path»[=[;«DPS»;][«expr»]]] */

/* ignore comment lines */
¢line  = Case( Left ( ¢line ; 1 ) ≠ "#" ; ¢line ) ;
¢isLineEmpty = IsEmpty( ¢line ) ;

/* split line at `=` and trim path horizontally right*/
¢posEquals = Position( ¢line ; "=" ; 1 ; 1 ) ;
$fmjaml__path = Case( not ¢posEquals ; ¢line ; While(
  [

  ¢len = ¢posEquals-1

  ]; Position ( ¢WS ; Middle( ¢line ; ¢len ; 1 ) ; 1 ; 1 ) ;[

  ¢len = ¢len - 1

  ];

  Left ( ¢line ; ¢len )

  ));

¢lineRest = Case( ¢posEquals ; Trim( Right( ¢line ; Length( ¢line ) - ¢posEquals ) ) ) ;

/* DPS */
¢posDps = Position ( ¢lineRest ; ";" ; 1 ; 1 ) ;
¢posDps = ¢posDps * IsEmpty( Substitute ( Left( ¢lineRest ; ¢posDps-1 ) ; [" " ; ""];[Char(9) ; ""] ) ) ;
¢posDpsEnd = Case( ¢posDps ; Position ( ¢lineRest ; ";" ; ¢posDps+1 ; 1 ) ) ;
¢dps = Case ( ¢posDpsEnd  ; Trim( Middle ( ¢lineRest ; 2 ; ¢posDpsEnd - 2 ) ) ) ;

/* expr inc. value_op */
¢expr  = Trim ( Right ( ¢lineRest ; Length ( ¢lineRest ) - ¢posDpsEnd ) ) ;
¢value_op = Case( 
Left ( ¢expr ; 1 ) ≠ "=" ; "" ; 
Left ( ¢expr ; 5 ) = "=====" ;  2 ; /* pos -> read block, 2*Quote() */
Left ( ¢expr ; 4 ) = "====" ; -2 ; /* neg -> read line,  2*Quote() */
Left ( ¢expr ; 3 ) = "===" ;  1 ; /* pos -> read block, 1*Quote() */
Left ( ¢expr ; 2 ) = "==" ; -1 ; /* neg -> read line,  1*Quote() */
0 /* pos -> read block, 0*Quote() */
);
$fmjaml__inc = 0 ;

/* DEBUG */
If=Case ( ¢debug ≥ 5 ; Let([
  $$fmjaml_log = List( $$fmjaml_log ;
  "line["&¢line_i&"] <- " & ¢line ;
  "- lineRest " & ¢lineRest ;
  Case( ¢debug ≥ 6 ; List ( 
    "- path:  " & Quote( $fmjaml__path ) ;
    Case ( not IsEmpty( ¢dps ) ; "- dps:  " & Quote( ¢dps ) );
    Case ( ¢posEquals ; "- value:  " & ¢expr ) ;
    ))
  ) ;
EndIf=0];0));

/* Process path array constructor */
$fmjaml__dtype = Case( Length( $fmjaml__path ) > 2 and Right( $fmjaml__path ; 2 ) = "[]" and
     not PatternCount( ¢TAB & " .[]" ; Middle ( $fmjaml__path ; Length( $fmjaml__path ) - 2 ; 1 ) ) ;
  Let( $fmjaml__path = Left ( $fmjaml__path ; Length( $fmjaml__path ) - 2 ) ;
  JSONArray )) ;

/**********
EXPAND PATH
***********

Support for avoiding key-repetition by inheriting empty keys from the previous line (ignoring white space):

For example:

Execute FileMaker Data API.target = "$result"
.request.layouts = "fmIDE Tests Light"
.       .limit = 1

becomes

Execute FileMaker Data API.target = "$result"
Execute FileMaker Data API.request.layouts = "fmIDE Tests Light"
Execute FileMaker Data API.request.limit = 1

Method:

1. ¢path_nows removes the white space
2. ¢path_dots also converts all delimiters to dots to simplify adjacent dot recognition.
3. Missing keys are inherited directly into $fmjaml_path (requiring ¢pos_path to be adjusted)
4. [+]/[:] is resolved in the resulting path before being used for the next line

Example @ line 3 above:

prev = `Execute FileMaker Data API.request.layouts`
path = `.       .limit`
nows = `..limit`
dots = `..limit`
*/

If = Case( not ¢isLineEmpty ; While ([

  ¢path_nows = Substitute ( $fmjaml__path ; [" " ; ""] ; [¢TAB ; ""] ) ;
  ¢path_dots = Substitute ( ¢path_nows ; ["[" ; "."] ; ["]" ; "."] ) ;

  ¢pos_path = 0 ;
  ¢pos_prev = 0 ;
  ¢pos_dot = 0 ;
  
  ¢pos_dot_next  = Position ( ¢path_dots ; "." ; ¢pos_dot + 1 ; 1 )
  
  ]; ¢pos_dot_next ; [
  
  /*

  1. Locate the next delimiter:
         v-¢pos_dot
  dots = "..limit"
          ^-¢pos_dot_next

  2. Identify delimiter:
          v-¢delim -> '.'
  nows = "..limit"

  3. Locate delimiter in path:
         v-¢pos_path
  path = ".       .limit"
          ^-¢pos_path_next

  4. Locate delimiter in prev-path:
         v-¢pos_prev
  prev = "Execute FileMaker Data API.request.layouts"
                                    ^-¢pos_prev_next
  key  = "**************************"

5. Inherit key from previous path only if dots adjacent

  */
  ¢delim         = Middle ( ¢path_nows ; ¢pos_dot_next ; 1 ) ;

  ¢pos_path_next = Position ( $fmjaml__path      ; ¢delim ; ¢pos_path + 1 ; 1 ) ;
  ¢pos_prev_next = Position ( $fmjaml__path_prev ; ¢delim ; ¢pos_prev + 1 ; 1 ) ;
  ¢pos_prev_next = Case ( ¢pos_prev_next ; ¢pos_prev_next ; Length ( $fmjaml__path_prev ) + 1 ) ;
  
  ¢inherit = ¢pos_dot_next = ¢pos_dot + 1 ;
  ¢key_prev = Case ( ¢inherit ; /* Get previous key */
    Middle ( $fmjaml__path_prev ; ¢pos_prev + 1 ; ¢pos_prev_next - ¢pos_prev - 1 )
  ) ;
  $fmjaml__path = Case ( not ¢inherit ; $fmjaml__path ; /* Replace empty key */
    Replace( $fmjaml__path      ; ¢pos_path + 1 ; ¢pos_path_next - ¢pos_path - 1 ; ¢key_prev )
  ) ;
  debug=Case ( ¢debug ≥ 7 and ¢inherit ; Let( $$fmjaml_log = List( $$fmjaml_log ;
  "¶Path Inheritance: " ;
  ¢line_i-1 & ":	" & Replace( Replace( $fmjaml__path_prev ; ¢pos_prev_next ; 1 ; "}" ) ; ¢pos_prev ; 1 ; "{" ) ;
  ¢line_i   & ":	" & Replace( Replace( $fmjaml__path      ; ¢pos_path_next ; 1 ; "}" ) ; ¢pos_path ; 1 ; "{" ) ;
  "key:	" & ¢key_prev ;
  "delim:	" & ¢delim ;
  "delims:	" & Replace( ¢path_nows ; ¢pos_dot_next ; 1 ; "•" )
  ) ; "")) ;

  /* 6. Make the next dot the current dot in dots + prev + path */
  ¢pos_dot  = ¢pos_dot_next ;
  ¢pos_prev = ¢pos_prev_next ;
  ¢pos_path = Case( ¢inherit ; ¢pos_path + Length ( ¢key_prev ) + 1 ;  ¢pos_path_next ) ;

  /* 7. Continue with the next dot */
  ¢pos_dot_next  = Position ( ¢path_dots ; "." ; ¢pos_dot + 1 ; 1 ) ;

EndIf=0];0)) ;


/**************
Support for [+] and [:] in FileMaker versions < 21
***************
*/

$fmjaml__path = Case( ¢fmp_v_maj ≥ 21 ; $fmjaml__path ; While ([

  ¢resolved_path = $fmjaml__path ;
  ¢pos_plus  = Position ( ¢resolved_path ; "[+]" ; 1 ; 1 ) ;
  ¢pos_colon = Position ( ¢resolved_path ; "[:]" ; 1 ; 1 ) 

  ] ; ¢pos_plus or ¢pos_colon ; [

  ¢pos = Case ( ¢pos_plus and ¢pos_colon ; Min ( ¢pos_plus ; ¢pos_colon ) ; ¢pos_plus + ¢pos_colon ) ;
  ¢array_length  = ValueCount ( JSONListKeys ( $fmjaml__json ; Left ( ¢resolved_path ; ¢pos - 1 ) ) ) ;
  ¢index = ¢array_length - ( ¢pos = ¢pos_colon ) ;
  ¢resolved_path = Replace ( ¢resolved_path ; ¢pos + 1 ; 1 ; Max( 0 ; ¢index ) ) ;
  ¢pos_plus  = Position ( ¢resolved_path ; "[+]" ; ¢pos+3 ; 1 ) ;
  ¢pos_colon = Position ( ¢resolved_path ; "[:]" ; ¢pos+3 ; 1 )

  ];

  ¢resolved_path

  )) ;

$fmjaml__path_prev = Case( ¢isLineEmpty ; $fmjaml__path_prev ; $fmjaml__path ) ;  

/******
heredoc
*******

Process value or heredoc
*/

$fmjaml__val = Case (
    ¢isLineEmpty ; "" ;

    /* no expression -> empty object */
    IsEmpty ( ¢expr ) ; "{}" ; Let(
      [
      /* Read value */
      ¢value = Case (
        IsEmpty( ¢value_op ) ; ¢expr ; 
        ¢value_op < 0 ; /* hereline */ Trim ( Replace( ¢expr ; 1 ; -¢value_op*2 ; "" ) ) ;
        /* 0 -> herecalc, >0 -> heredoc */ Let(
          [ 
          ¢marker    = Replace ( "==" & ¢expr ; 1 ; ¢value_op*2 ; "" ) ;/*
"=…" → 0 → +== → "===…" 
"===…" → 1 → +==-== → "===…" 
"=====…" → 2 → +==-==== → "===…" 
*/
          ¢pos_start = Length( LeftValues ( ¢lines ; ¢line_i ) )+1 ;
          ¢pos_end   = Position ( ¢lines ; ¢marker ; ¢pos_start ; 1 ) ;
          ¢heredoc   = Middle( ¢lines ; ¢pos_start ; ¢pos_end - ¢pos_start /* and remove last EOL: */ -1 ) ;
          $fmjaml__inc = ValueCount( ¢heredoc )+1
          ];
          ¢heredoc
          )
        )

      ];

      /* Return value, quoting depending on the operator */
      Choose( Abs ( ¢value_op ) ;
        Evaluate ( ¢value ) ;
        ¢value ;
        Quote( ¢value )
      )
      
      )    
    );


/*
Calculate the line-increment value based on the number of lines consumed by the heredoc.
This is required for the next section where the DPS needs to drop the key if an empty condition (=EOF) is reached
*/
¢line_inc     = 1 + $fmjaml__inc ;


/******
;«DPS»;
*******

Process ;«DPS»;
*/

$fmjaml__sticky_op = "" ;
$fmjaml__context = "" ;
$fmjaml__clause_empty = "" ;
$fmjaml__present = "" ;
$fmjaml__test = "" ;
$fmjaml__skip = "" ;
¢param = FilterValues( Trim ( Substitute ( ¢dps ; "==" ; "") ) ; "release¶release_tag¶version" ) ;
¢r = Case( not IsEmpty ( ¢param ) ; Let([ 
  ¢r = Substitute( RightWords ( ¢release ; Case( ¢param = "release¶" ; 99 ; 1 ) ) ; "v" ; Case( ¢param = "release_tag¶" ; "v" ) ) ;
  $fmjaml__val = Case ( Position ( ¢dps ; "==" ; 1 ; 1 ) ; Quote ( ¢r )  ; ¢r ) ;
  $fmjaml__dtype = JSONString
  ];0) ;
  not IsEmpty ( ¢dps ) ; While (
  [

  /*************
  FOR EACH ¢char
  **************

  in the ¢dps, at least until dropped…
  */

  ¢dpsPos = 1 ;
  ¢dpsLength = Length( ¢dps )

  ]; ¢dpsPos <= ¢dpsLength+1 and not IsEmpty ( $fmjaml__path );[

  ¢char = Middle ( ¢dps ; ¢dpsPos ; 1 ) ;
  $fmjaml__chars_read = 1 ;

  ¢eof = ¢dpsPos > ¢dpsLength ;
  ¢eoClause = ¢eof or Position ( "?-:," ; ¢char ; 1 ; 1 ) ;

  /* DEBUG fmJAML */
  If=Case ( ¢debug ≥ 7 ; Let([
      $$fmjaml_log = List( $$fmjaml_log ;
      "DPS[" & ¢dpsPos & "]: "  & Quote ( ¢char ) & Case( ¢eof ; " («EOF»)" );
      "- $fmjaml__dtype: "        & $fmjaml__dtype ;
      "- $fmjaml__present: "      & Quote ( $fmjaml__present ) ;
      "- $fmjaml__test: "         & Quote ( $fmjaml__test ) ;
      "- $fmjaml__skip: "         & Quote ( $fmjaml__skip ) ;
      "- $fmjaml__clause_empty: " & Quote ( $fmjaml__clause_empty ) ;
      "- $fmjaml__sticky_op: "    & Quote ( $fmjaml__sticky_op ) ;
      "- $fmjaml__context: "      & Quote ( $fmjaml__context ) ;
      "- $fmjaml__path: "         & Quote ( $fmjaml__path ) ;
      "- $fmjaml__val: "          & Quote ( $fmjaml__val ) ;
      ) ;
  EndIf=0];0)) ;

  /*****************
  Lazy data coercion
  ******************

  Convert the value to the target data-type, but first when a processing operator (`!`/`?`/`-`) is encountered
TNBAOUJDIMR
  */
  ¢notIE = not IsEmpty ( $fmjaml__val ) ;
  ¢r = Case(
    ¢eof or IsEmpty ( $fmjaml__dtype ) ; "" ;
    Position ( "!?-" ; ¢char ; 1 ; 1 ) = 0 ; "" ;
    Let( $fmjaml__val = Case( 
    $fmjaml__dtype = JSONString     ; GetAsText      ( $fmjaml__val ) ;
    $fmjaml__dtype = JSONNumber     ; GetAsNumber    ( $fmjaml__val ) ;
    $fmjaml__dtype = JSONObject     ; Case( ¢notIE ; JSONGetElement ( $fmjaml__val ; "" ) ) ;
    $fmjaml__dtype = JSONArray      ; Case( ¢notIE ; JSONGetElement ( $fmjaml__val ; "" ) ) ;
    $fmjaml__dtype = JSONBoolean    ; GetAsBoolean   ( $fmjaml__val ) ;
    $fmjaml__dtype = JSONNull       ; ""                              ;
    $fmjaml__dtype = JSONRaw        ; Case( ¢notIE ; JSONGetElement ( $fmjaml__val ; "" ) ) ;
    $fmjaml__dtype = ¢JSONDate      ; GetAsDate      ( $fmjaml__val ) ;
    $fmjaml__dtype = ¢JSONTime      ; GetAsTime      ( $fmjaml__val ) ;
    $fmjaml__dtype = ¢JSONTimestamp ; GetAsTimeStamp ( $fmjaml__val ) ;
    /* no coercion to container */
    $fmjaml__val
    ) ; "" )
  );

  /*************
  Parse operator
  **************

  Switch statement to process the encountered operator
  */
  ¢parseNextOp = Case(

    /* SWITCH ( ¢char ) */


    /* Lazily drop path, if necessary, which is when… */
    ¢eoClause and $fmjaml__clause_empty and /* …completed, empty (or missing) clause and… */ 
    Case( $fmjaml__present ;
      /* …value present and … explicitly empty presence clause: */ 
      $fmjaml__context = "?" and ¢char = ":"    /* value + "?💢:" -> */
      or
      $fmjaml__context = "-:" ;                 /* value + "-:💢" -> */

      /* …value empty and… */
      $fmjaml__context = "?" and ¢char ≠ ":"    /* empty + "?💢" (implicit) -> */
      or
      $fmjaml__context = "?:"                   /* empty + "?:💢" (explicit) -> */
    )
    ; Let( $fmjaml__path = "" ; "" ) ;          /* -> drop value + path! */ 

    /* Ignore EOF or white space */

    ¢eof or
    ¢char = " " or
    ¢char = ¢TAB ; "" ;

    /* Process end-skip chars (`:` / `,`) *before* processing any skipping */

    ¢char = "," ; Let ([/* End conditional scope */
                        $fmjaml__present = "" ;
                        $fmjaml__test = "" ;
                        $fmjaml__skip = "" ;
                        $fmjaml__clause_empty = "" ;
                        $fmjaml__context = "" ;
                        $fmjaml__sticky_op = ""
                       ]; "" ) ;
    ¢char = ":" ; Let ([/* otherwise clause */
                        $fmjaml__context = Case(
                          $fmjaml__context = "?" ; "?:" ;
                          $fmjaml__context = "-" ; "-:" ;
                          $fmjaml__context
                          ) ;
                        $fmjaml__skip = $fmjaml__test ;
                        $fmjaml__clause_empty  = 1 ;
                        $fmjaml__sticky_op = ""
                       ]  ; "" ) ;

    /*****************
    'value' or ~regex~
    ******************

    Read the entire apostrophe-string, then either 
    - if skipping             -> ignore it, otherwise (mark the current clause as not empty and…)
    - if not-value (`!'…'`)   -> remove the value if present or
    - if EOL op (`¶'…'`)      -> replace EOLs with value
    - if divide op (`/`)      -> divide value
    - if replace op (`>`)     -> (regex-)replace value
    - if normal value (`'…'`) -> set the value to the string.
    */
    ¢char = "`" or
    ¢char = "'" or
    ¢mbs and /* if mbs is not present just ignore the regex op -> unknown char error below */
    ¢char = "~" ; Let([¢isRegex = ¢char = "~" ;
                       $fmjaml__string_char = ¢char;
                       ¢dpsValueEnd = Position ( ¢dps & ¢char ; ¢char ; ¢dpsPos+1 ; 1 ) ;
                       $fmjaml__chars_read = ¢dpsValueEnd+1 - ¢dpsPos ;

                       /* Interpret a pilcrow `\¶` as an EOL `¶`  */
                       ¢dpsValue = Substitute( Middle( ¢dps ; ¢dpsPos+1 ; $fmjaml__chars_read - 2 ) ; "\¶" ; ¶ ) ;
                       ¢dpsValue = Case( ¢char = "`" ; Evaluate ( ¢dpsValue ) ; ¢dpsValue ) ;

                       ¢dpsGTPos = Position ( ¢dps ; ">" ; ¢dpsValueEnd+1 ; 1 ) ;
                       ¢dpsReplaceOp = ¢dpsGTPos and (¢dpsGTPos = ¢dpsValueEnd+1 or  IsEmpty ( Substitute ( Middle ( ¢dps ; ¢dpsValueEnd+1 ; ¢dpsGTPos - ¢dpsValueEnd - 1 ) ; [" ";""] ; [Char(9);""] ) ) ) ;

                       If=Case ( ¢debug ≥ 8 ; Let([
                           $$fmjaml_log = List ( $$fmjaml_log ;
                           "¢char=" & ¢char ;
                           "$fmjaml__sticky_op: " & $fmjaml__sticky_op ;
                           "$fmjaml__skip: " & $fmjaml__skip ;
                           "¢dpsGTPos: " & ¢dpsGTPos ;
                           "¢dpsReplaceOp: " & ¢dpsReplaceOp ;
                           ) ;
                       EndIf=0];"")) ;

                       /* Ignore the value if skipping…  */
                        If=Case ( not $fmjaml__skip ; Let([

                            /* …else mark clause as not empty and process the value… */
                            $fmjaml__clause_empty = "" ;
                            $fmjaml__val = Case(

                              /* Not op `!…`*/
                              $fmjaml__sticky_op = "!" ; Case ( Case ( ¢isRegex ;
                                  /* If regex match… */
                                  Let([$fmjaml__eval = "MBS(\"RegEx.Match\";" & Quote ( ¢dpsValue ) & ";" & Quote ( $fmjaml__val ) & ")"
                                      ]; Exact( Evaluate ( $fmjaml__eval ) ; 1 )
                                      ) ;
                                  /* If value is equal… (FIXME exact?) */
                                  $fmjaml__val = ¢dpsValue
                                ) /* …REMOVE, otherwise keep */
                                  ; "" ; $fmjaml__val ) ;

                              /* EOL op `¶'…'` -> Replace EOLs */
                              $fmjaml__sticky_op = "¶" ; Let([
                                $fmjaml__sticky_op = "" ;
                                vResult = Substitute ( $fmjaml__val
                                                     ; [Char(13) & Char(10);¶]       /* CRLF -> CR */
                                                     ; [Char(10);¶]                  /* LF -> CR */
                                                     ; [¶ ; ¢dpsValue] /* CR -> replacement */
                                                     )
                                ];
                                vResult
                                ) ;

                              /* Divide op  (/'…') */
                              $fmjaml__sticky_op = "/" ; Let( $fmjaml__sticky_op = "" ; $fmjaml__val / ¢dpsValue ) ;



                              /* replace operation  ('…'>'…') -> Replace the value by the text */
                              $fmjaml__sticky_op = ">" ; Let([
                                $fmjaml__sticky_op = "" ;
                                vResult = Case( ¢isRegex ;
                                    /* Regex replace */
                                  Let([$fmjaml__eval = "MBS(\"RegEx.ReplaceAll\";" & Quote ( $fmjaml__val ) & ";" & Quote ( $fmjaml__replace ) & ";" & Quote ( ¢dpsValue ) & ";" & Quote ( $fmjaml__attributes ) & ")"
                                      ]; Evaluate ( $fmjaml__eval )
                                      ) ;
                                    /* Substitute replace */
                                    Substitute ( $fmjaml__val ; $fmjaml__replace ; ¢dpsValue )
                                  );
                                $fmjaml__replace = ""
                                ];
                                vResult
                                ) ;


                              /* replace operation  ('…'>'…') -> Remember attributes */
                              $fmjaml__sticky_op = "@" ; Let([
                                $fmjaml__sticky_op = "" ;
                                $fmjaml__attributes = ¢dpsValue
                                ];
                                $fmjaml__val /* don't change */
                                ) ;

                              /* else if followed by a `>` -> just remember value to replace ('…'>) and replaceOp */
                              ¢dpsReplaceOp ; Let([
                                $fmjaml__replace = ¢dpsValue ;
                                $fmjaml__replaceOp = ¢char
                                ];
                                $fmjaml__val /* don't change */
                                ) ;

                              /* else set-text operation ('…') -> Set value */
                                ¢dpsValue
                              );

                        EndIf=0];0))
                      ]; "" ) ;

    /* Process skip: skip any other simple operators (and continue skipping up to any of the above operators or <EOF>) */

    $fmjaml__skip ; Let([ /* DEBUG fmJAML */
                        If=Case ( ¢debug ≥ 8 ; Let([
                          $$fmjaml_log = List( $$fmjaml_log ;
                                         "SKIPPED: '" & ¢char & "'" ;
                                         ) ;
                        EndIf=0];""))
                      ];"") ;

    /* ANY other active character indicates the clause is not empty */
    $fmjaml__clause_empty and Let( $fmjaml__clause_empty = "" ; False ) ; "" ;


    /* Perform the next operator (using Choose for speed)… */
    Choose ( Position( "TNOABUJDIMR*!\¶?/" ; Substitute( ¢char ; ["-" ; "?"]; [">" ; "/"]; ["@" ; "/"]; ["#" ; "/"] ) ; 1 ; 1 ) ;

    /* ignore unknown char */ "" ; 

    /* Process data type */

    /* "T" ;*/ Let( $fmjaml__dtype = JSONString     ; "" ) ;
    /* "N" ;*/ Let( $fmjaml__dtype = JSONNumber     ; "" ) ;
    /* "O" ;*/ Let( $fmjaml__dtype = JSONObject     ; "" ) ;
    /* "A" ;*/ Let( $fmjaml__dtype = JSONArray      ; "" ) ;
    /* "B" ;*/ Let( $fmjaml__dtype = JSONBoolean    ; "" ) ;
