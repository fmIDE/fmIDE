
# This script demonstrates the fmIDE action given by the script's name
# A single action object is passed to fmIDE

Set Variable [ $action_name[1] ; Get( ScriptName ) ]
Perform Script [ Specified: By name ; "fmIDE" ; Parameter: JSONSetElement ( "" ; "action_name" ; $action_name ; JSONString ) ]
Set Variable [ $version[1] ; JSONGetElement( Get( ScriptResult ) ; "data.version" ) ]

Show Custom Dialog [ Title: Get( FileName ) ; Message: Get( ScriptResult ) ; Default Button: "OK" ]
Show Custom Dialog [ Title: Get( FileName ) ; Message: "Version: " & $version ; Default Button: "OK" ]
