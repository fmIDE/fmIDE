
# This script demonstrates the fmIDE action given by the script's name
# A single action object is passed to fmIDE

Set Variable [ $table_name[1] ; "Example" ]
Perform Script [ Specified: By name ; "fmIDE" ; Parameter: JSONSetElement ( "{}" 
				; ["Get Field Names.table_name" ; $table_name ; JSONString ]
				) ]
Set Variable [ $_JSR[1] ; Case( not IsEmpty ( Get( ScriptResult ) ) ; Get( ScriptResult ) ; "{}" ) ]
Set Variable [ $field_names[1] ; JSONGetElement( $_JSR ; "data.field_names" ) ]

Show Custom Dialog [ Message: $field_names ; Default Button: "OK" ]
