# Passing a table_name performs a Go to Table action
# Note: Because the Go to Table action changes layout in order to target the correct table, you have to go to the original layout afterwards

Set Variable [ $table_name[1] ; Case ( Get ( ActiveModifierKeys ) ; "_Table2" ; "_Table1" ) ]
Perform Script [ Specified: By name ; "fmIDE" ; Parameter: JSONSetElement ( "" ; "table_name" ; $table_name ; JSONString ) ]

Go to Layout [ Layout: <original layout> ; Animation: None ]
