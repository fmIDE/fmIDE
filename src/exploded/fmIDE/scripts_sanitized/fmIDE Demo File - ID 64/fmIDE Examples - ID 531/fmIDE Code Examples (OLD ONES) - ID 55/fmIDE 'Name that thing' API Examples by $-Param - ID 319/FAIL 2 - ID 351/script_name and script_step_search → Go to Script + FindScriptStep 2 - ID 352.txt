# Passing a script_name performs a Go to Script action

Set Variable [ $script_name[1] ; "fmIDE" ]
Set Variable [ $script_step_number[1] ; 100 ]
Set Variable [ $script_step_search[1] ; "$action_name = FindScriptStep" ]
Perform Script [ Specified: By name ; "fmIDE" ; Parameter: JSONSetElement ( "{}" ; 
				["script_name" ; $script_name ; JSONString] ;
				["script_step_number" ; $script_step_number ; JSONNumber] ;
				["script_step_search" ; $script_step_search ; JSONString]
				) ]
