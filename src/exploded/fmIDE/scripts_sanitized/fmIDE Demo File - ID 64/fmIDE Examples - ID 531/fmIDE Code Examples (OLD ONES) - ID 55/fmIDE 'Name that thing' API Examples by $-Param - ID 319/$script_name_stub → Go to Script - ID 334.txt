# Passing script_name_stub runs the first script starting with that prefix
# Thus this script manages to find itself - even though only the first part of the name is specified!
# Cool! 😎

Set Variable [ $script_name_stub[1] ; "$script_name_stub" ]
Set Variable [ $script_step_number[1] ; 3 ]
Perform Script [ Specified: By name ; "fmIDE" ; Parameter: "$script_name_stub=" & Quote( $script_name_stub ) & ";" &
				"$script_step_number=" & Quote( $script_step_number ) & ";" ]
