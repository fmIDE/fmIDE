
# This script demonstrates the fmIDE action given by the script's name
# A single action object is passed to fmIDE

Set Variable [ $action_name[1] ; Get( ScriptName ) ]
Set Variable [ $custom_menu_name[1] ; "File Menu" ]
Perform Script [ Specified: By name ; "fmIDE" ; Parameter: JSONSetElement ( "" ;
				["action.action_name" ; $action_name ; JSONString] ;
				["action.custom_menu_name" ; $custom_menu_name ; JSONString]
				) ]
