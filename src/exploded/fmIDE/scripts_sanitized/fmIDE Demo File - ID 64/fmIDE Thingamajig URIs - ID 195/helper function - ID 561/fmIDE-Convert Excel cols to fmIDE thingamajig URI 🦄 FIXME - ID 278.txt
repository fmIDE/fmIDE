
# 20221005 Convert active field contents from tab delimited columns to thingamajig uris

Set Variable [ $text[1] ; Get ( ActiveFieldContents ) ]
Insert Calculated Result [ Select: ON ; While ( 
				[
				¢text = $text ;
				
				¢result = "" ;
				
				/* ANALYSE FIELD HEADERS */
				¢headers = Substitute( GetValue ( ¢text ; 1 ) ; Char( 9 ) ; "¶" ) ;
				¢header_count = ValueCount( ¢headers ) ;
				
				¢analyse_headers_and_create_vars = 
				While ( 
				[
				¢j = 2;
				¢m = ¢header_count
				
				]; ¢j <= ¢m ;[
				
				¢header = GetValue( ¢headers ; ¢j ) ;
				$$fmide_thigamajig_header[¢j] = ¢header ;
				
				¢is_colons = Position( ¢header ; "::" ; 1 ; 1 ) > 0 ;
				¢header_bits = Substitute( ¢header ; "::" ; "¶" ) ;
				¢table_name = Case ( ¢is_colons ; GetValue( ¢header_bits ; 1 ) ) ;
				¢field_name = GetValue( ¢header_bits ; 1 + ¢is_colons ) ;
				
				
				$$fmide_thing[¢j] = Case( 
				
				/* Base elements */
				¢field_name = "cPosition Description" ; "script_step" ;
				¢field_name = "ParentDesc1" ; "type : name" ;
				¢field_name = "ParentDesc2" ; "type : name" ;
				
				
				/* Best guess columns */
				Position( ¢field_name ; "data"  ; 1 ; 1 ) ; "external_data_source" ;
				Position( ¢field_name ; "src"   ; 1 ; 1 ) ; "external_data_source" ;
				Position( ¢field_name ; "ref"   ; 1 ; 1 ) ; "external_data_source" ;
				
				Position( ¢field_name ; "func"  ; 1 ; 1 ) ; "account" ;
				Position( ¢field_name ; "fct"   ; 1 ; 1 ) ; "account" ;
				
				Position( ¢field_name ; "func"  ; 1 ; 1 ) ; "custom_function" ;
				Position( ¢field_name ; "fct"   ; 1 ; 1 ) ; "custom_function" ;
				Position( ¢field_name ; "cf"   ; 1 ; 1 ) ; "custom_function" ;
				
				Position( ¢field_name ; "base"  ; 1 ; 1 ) ; "base_table" ;
				Position( ¢field_name ; "tbl"   ; 1 ; 1 ) ; "base_table" ;
				Position( ¢field_name ; "table" ; 1 ; 1 ) ; "base_table" ;
				Position( ¢field_name ; "tbl"   ; 1 ; 1 ) ; "base_table" ;
				Position( ¢field_name ; "field" ; 1 ; 1 ) ; "field" ;
				Position( ¢field_name ; "fld"   ; 1 ; 1 ) ; "field" ;
				
				Position( ¢field_name ; "val"   ; 1 ; 1 ) ; "value_list" ;
				Position( ¢field_name ; "vl"    ; 1 ; 1 ) ; "value_list" ;
				
				Position( ¢field_name ; "obj"   ; 1 ; 1 ) ; "object" ;
				Position( ¢field_name ; "lay"   ; 1 ; 1 ) ; "layout" ;
				Position( ¢field_name ; "lyt"   ; 1 ; 1 ) ; "layout" ;
				
				Position( ¢field_name ; "Step"  ; 1 ; 1 ) ; "script_step" ;
				Position( ¢field_name ; "Scr"   ; 1 ; 1 ) ; "script" ;
				
				Position( ¢field_name ; "ext"   ; 1 ; 1 ) and 
				Position( ¢field_name ; "priv"  ; 1 ; 1 ) ; "external_privilege" ;
				Position( ¢field_name ; "priv"  ; 1 ; 1 ) ; "privilege_set" ;
				
				"unknown"
				);
				
				
				$$fmide_id_type[¢j] = Case( 
				$$fmide_thing[¢j] = "script_step" or
				$$fmide_thing[¢j] = "menu_item" or
				Position( ¢field_name ; "num"  ; 1 ; 1 ) ; "number" ;
				Position( ¢field_name ; "nam"   ; 1 ; 1 ) ; "name" ;
				Position( ¢field_name ; "uuid"   ; 1 ; 1 ) ; "uuid" ;
				Position( ¢field_name ; "id"   ; 1 ; 1 ) ; "id" ;
				"?"
				);
				
				
				¢j = ¢j + 1
				
				];
				
				""
				
				);
				
				
				
				
				/* PROCESS DATA */
				¢i = 2;
				¢n = ValueCount ( ¢text )
				
				]; ¢i <= ¢n ;[
				
				¢line = GetValue( ¢text ; ¢i ) ;
				
				¢result = ¢result & 
				While ( 
				[
				¢cells = Substitute( ¢line ; Char(9) ; "¶" ) ;
				
				/* first the file_name */
				¢line_result = Substitute( GetValue( ¢cells ; 1 ) ; ".fmp12" ; "" ) ;
				
				¢j = 2;
				¢m = ¢header_count
				
				]; ¢j <= ¢m ;[
				
				¢cell = GetValue( ¢cells ; ¢j ) ;
				
				¢line_result = ¢line_result & Case( 
				
				  /* multiple-type column (Base elements calcs) */
				  $$fmide_thing[¢j] = "type : name" ; 
				  Let([
				  ¢bits = Substitute( ¢cell ; " : " ; "¶" ) ;
				  ¢cell = GetValue ( ¢bits ; 2 ) ;
				  ¢thing = GetValue ( ¢bits ; 1 ) ;
				  ¢thing = Substitute( Trim( Substitute( ¢thing ; 
				["A"; " a"];
				["B"; " b"];
				["C"; " c"];
				["D"; " d"];
				["E"; " e"];
				["F"; " f"];
				["G"; " g"];
				["H"; " h"];
				["I"; " i"];
				["J"; " j"];
				["K"; " k"];
				["L"; " l"];
				["M"; " m"];
				["N"; " n"];
				["O"; " o"];
				["P"; " p"];
				["Q"; " q"];
				["R"; " r"];
				["S"; " s"];
				["T"; " t"];
				["U"; " u"];
				["V"; " v"];
				["W"; " w"];
				["X"; " x"];
				["Y"; " y"];
				["Z"; " z"]
				) ) ;
				["  "; " "];
				[" "; "_"]
				);
				  ¢thing = Substitute( ¢thing ; ["step"; "script_step"] ) ;
				
				
				
				  ¢type = Case( ¢thing = "script_step" or ¢thing = "menu_item" ; "number" ; "name" )
				  ];
				  "&$" & ¢thing & "_" & ¢type & "=" & GetAsURLEncoded( ¢cell ) 
				  );
				
				  /* single-type column */
				  "&$" & $$fmide_thing[¢j] & "_" & $$fmide_id_type[¢j] & "=" & GetAsURLEncoded( ¢cell ) 
				  );
				
				¢j = ¢j + 1
				
				];
				
				¢line_result
				
				) & "¶";
				
				¢i = ¢i + 1
				
				];
				
				¢result
				
				) ]
