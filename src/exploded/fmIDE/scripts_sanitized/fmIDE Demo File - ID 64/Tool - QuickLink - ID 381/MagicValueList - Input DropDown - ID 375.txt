
# 2025-09-25 RW: new

Set Error Capture [ ON ]
Allow User Abort [ OFF ]

Set Variable [ $_JSP[1] ; _JSP ]
Set Variable [ $prompt[1] ; JSONGetElement ( $_JSP ; "prompt" ) ]
Set Variable [ $title[1] ; JSONGetElement ( $_JSP ; "title" ) ]
Set Variable [ $value[1] ; JSONGetElement ( $_JSP ; "value" ) ]
Set Variable [ $magic_values[1] ; JSONGetElement ( $_JSP ; "magic_values" ) ]

Set Variable [ $title[1] ; _Default( $title ; Get( FileName ) ) ]

Set Field [ fmIDE::_gMVL_Title ; $title ]
Set Field [ fmIDE::_gMVL_Prompt ; $prompt ]
Set Field [ fmIDE::_gMVL_Input ; $value ]

Set Variable [ $WindowName[1] ; $title ]
Set Variable [ $WindowWidth[1] ; 770 ]
Set Variable [ $WindowHeight[1] ; 210 ]
Set Variable [ $WindowLeft[1] ; Div( Get ( WindowDesktopWidth ) - $WindowWidth ; 2 ) ]
Set Variable [ $WindowTop[1] ; Div( Get ( WindowDesktopHeight ) - $WindowHeight ; 2 ) ]

New Window [ Style: Dialog ; Name: $WindowName ; Layout: "AC MagicValueList DropDown" ; Height: $WindowHeight ; Width: $WindowWidth ; Top: $WindowTop ; Left: $WindowLeft ; Close: OFF ; Minimize: OFF ; Maximize: OFF ; Resize: OFF ; Menu: OFF ; Toolbar: OFF ]

Set Variable [ $$MVL_DropDown[1] ; $magic_values ]

Go to Field [ fmIDE::_gMVL_Input ]
Pause/Resume Script [ ⚠️ PARAMETER "Options" NOT PARSED ⚠️ ]

If [ $$LastMessageChoice = 1 ]
	# OK
	Set Variable [ $value[1] ; fmIDE::_gMVL_Input ]
Else
	# Cancel
	# Don't read
End If

Close Window

Exit Script [ $value ]
