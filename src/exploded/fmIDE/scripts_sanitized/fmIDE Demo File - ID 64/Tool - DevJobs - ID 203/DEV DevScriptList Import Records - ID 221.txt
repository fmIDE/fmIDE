Go to Layout [ Layout: "🚨🚨🚨 <BROKEN REFERENCE> 🚨🚨🚨" ; Animation: None ]

	Show Custom Dialog [ Title: Get( ScriptName ) ; Message: "Fehler " & Get( LastError ) ; Default Button: "OK" ]

	Perform Script [ "" ; Specified: From list ; Parameter: Get( ScriptName ) ]

Go to Layout [ Layout: "🚨🚨🚨 <BROKEN REFERENCE> 🚨🚨🚨" ; Animation: None ]
Replace Field Contents [ fmDevJobTask::__UUID ; Replace with calculation: $DevJobID ]

	Go to Layout [ Layout: "🚨🚨🚨 <BROKEN REFERENCE> 🚨🚨🚨" ; Animation: None ]
	Go to Field [ fmDevJob::🚨🚨🚨 <BROKEN REFERENCE> 🚨🚨🚨 ]
Else
	Set Variable [ $FC[1] ; Get( FoundCount ) ]
	Enter Find Mode [ Pause: OFF ]
	Set Field [ fmDevJobTask::__UUID ; $DevJobID ]
	Perform Find
	If [ /*G_VL_Found_OK = "OK"*/ ]
		Go to Record/Request/Page [ With dialog: OFF ; Get( FoundCount ) - $FC + 1 ]
		Go to Field [ fmDevJobTask::Task ]
	End If
