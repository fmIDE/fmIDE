
# 20221004 MrW Fixed bug where wrong line would be read when the cursor is at the end of the line

Set Error Capture [ ON ]

Set Variable [ $thingamajig[1] ; Let ( 
				[
				
				¢ActiveFieldContents   = Get ( ActiveFieldContents ) ;
				¢ActiveSelectionStart  = Get ( ActiveSelectionStart ) ;
				¢ActiveSelectionSize   = Get ( ActiveSelectionSize ) ;
				
				¢Line = Case (
				
				    /* If something is selected get selected text */
				    ¢ActiveSelectionSize ; Middle ( ¢ActiveFieldContents ; ¢ActiveSelectionStart ; ¢ActiveSelectionSize ) ; 
				
				    /* Otherwise get line under cursor */
				    /* Note: If the cursor is at the end of the line, don't count the EOL to the right of the cursor */
				    Let(¢SelectionLineNr = PatternCount ( Left ( ¢ActiveFieldContents ; ¢ActiveSelectionStart - Case (¢ActiveSelectionSize=0 and Middle ( ¢ActiveFieldContents ; ¢ActiveSelectionStart ; 1 )="¶" ; 1 ) ) ; "¶" ) +1 ;
				        GetValue( ¢ActiveFieldContents ; ¢SelectionLineNr )
				    )
				  );
				
				
				/* Remove leading Text */
				¢PosParamsStart = Position ( ¢Line ; "&" ; 1 ; 1 ) ;
				¢PosLastSpace = Position ( ¢Line ; " " ; ¢PosParamsStart ; -1 ) ;
				¢Thingamajig = Case( ¢PosLastSpace ; Right ( ¢Line ; Length ( ¢Line ) - ¢PosLastSpace ) ; ¢Line )
				
				///* Debug: */;$$Thingamajig = ¢Thingamajig & "/" & ¢PosParamsStart & "/" & ¢PosLastSpace
				
				];
				
				¢Thingamajig
				
				
				
				) ]
