# Use this script in Install Timer Script calls, in order to avoid hard-coding a call to fmIDE 


# @var $file_name
# @var $script_name
# @var $parameter
# @var $after_seconds

Set Variable [ $_JSP[1] ; Case ( not IsEmpty ( Get ( ScriptParameter ) ) ; Get ( ScriptParameter ) ; "{}" ) ]
Set Variable [ $file_name[1] ; Case ( not IsEmpty ( $file_name ) ; $file_name ; JSONGetElement ( $_JSP ; "file_name" )) ]
Set Variable [ $script_name[1] ; Case ( not IsEmpty ( $script_name ) ; $script_name ; JSONGetElement ( $_JSP ; "script_name" )) ]
Set Variable [ $parameter[1] ; Case ( not IsEmpty ( $parameter ) ; $parameter ; JSONGetElement ( $_JSP ; "parameter" )) ]
Set Variable [ $after_seconds[1] ; Case ( not IsEmpty ( $after_seconds ) ; $after_seconds ; JSONGetElement ( $_JSP ; "after_seconds" )) ]
Set Variable [ $after_seconds[1] ; Case ( not IsEmpty ( $after_seconds ) ; $after_seconds ; 1/100 ) ]


Set Variable [ $$fmIDE_RunOnTimerScriptOnce_closure[1] ; JSONSetElement ( "{}"
				; ["file_name" ; $file_name ; JSONString]
				; ["script_name" ; $script_name ; JSONString]
				; ["parameter" ; $parameter ; JSONString]
				) ]
Install OnTimer Script [ "fmIDE-RunOnTimerScriptOnce private" ; $after_seconds ]
